// videocall.js
const socket = io("http://localhost:3000"); // signaling server

const localVideo = document.getElementById("localVideo");
const remoteVideo = document.getElementById("remoteVideo");
const startCallBtn = document.getElementById("startCallBtn");
const hangupBtn = document.getElementById("hangupBtn");
const callStatus = document.getElementById("callStatus");
const roomIdDisplay = document.getElementById("roomIdDisplay");

let localStream;
let peerConnection;
const configuration = { iceServers: [{ urls: "stun:stun.l.google.com:19302" }] };
let roomId = "braintrade-room";

roomIdDisplay.textContent = roomId;

startCallBtn.onclick = async () => {
  localStream = await navigator.mediaDevices.getUserMedia({ video: true, audio: true });
  localVideo.srcObject = localStream;

  peerConnection = new RTCPeerConnection(configuration);

  localStream.getTracks().forEach(track => peerConnection.addTrack(track, localStream));

  peerConnection.ontrack = e => {
    remoteVideo.srcObject = e.streams[0];
  };

  peerConnection.onicecandidate = e => {
    if (e.candidate) {
      socket.emit("signal", { room: roomId, signal: e.candidate });
    }
  };

  socket.emit("join", roomId);

  socket.on("signal", async data => {
    if (data.signal.candidate) {
      await peerConnection.addIceCandidate(new RTCIceCandidate(data.signal));
    } else if (data.signal.sdp) {
      await peerConnection.setRemoteDescription(new RTCSessionDescription(data.signal));
      if (data.signal.type === "offer") {
        const answer = await peerConnection.createAnswer();
        await peerConnection.setLocalDescription(answer);
        socket.emit("signal", { room: roomId, signal: answer });
      }
    }
  });

  const offer = await peerConnection.createOffer();
  await peerConnection.setLocalDescription(offer);
  socket.emit("signal", { room: roomId, signal: offer });

  callStatus.textContent = "Call started";
  startCallBtn.disabled = true;
  hangupBtn.disabled = false;
};

hangupBtn.onclick = () => {
  peerConnection.close();
  peerConnection = null;
  localVideo.srcObject = null;
  remoteVideo.srcObject = null;
  callStatus.textContent = "Call ended";
  startCallBtn.disabled = false;
  hangupBtn.disabled = true;
  socket.emit("leave", roomId);
};
